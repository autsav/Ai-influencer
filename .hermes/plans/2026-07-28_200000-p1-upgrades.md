"""Multi-reference face loader: rotates through reference face images."""
import os
import logging
from pathlib import Path

log = logging.getLogger(__name__)

def load_reference_faces(ref_dir: str) -> list[bytes]:
    """Load all .jpg/.png files from the reference faces directory."""
    p = Path(ref_dir)
    if not p.exists():
        return []
    faces = []
    for f in sorted(p.glob("*")):
        if f.suffix.lower() in (".jpg", ".jpeg", ".png", ".webp"):
            faces.append(f.read_bytes())
    log.info("Loaded %d reference faces from %s", len(faces), ref_dir)
    return faces

def select_reference_face(faces: list[bytes], seed: int | None = None) -> bytes | None:
    """Select a reference face by seed (deterministic rotation) or random."""
    if not faces:
        return None
    if seed is not None:
        return faces[seed % len(faces)]
    import random
    return random.choice(faces)
```

# aeloria/budget.py — full rewrite
import logging
import httpx
from aeloria.config import Settings
from aeloria.db.client import Db

log = logging.getLogger(__name__)

class BudgetExceeded(Exception):
    pass

def check_budget(db: Db, settings: Settings, engine: str, planned_cost: float) -> None:
    if engine == "fal":
        daily_cap = settings.fal_daily_cap
        weekly_cap = settings.fal_weekly_cap
    elif engine == "higgsfield":
        daily_cap = settings.higgsfield_daily_credits_cap
        weekly_cap = float("inf")
    else:
        raise ValueError(f"unknown engine: {engine}")

    spent_today = db.sum_media_cost_today(engine) or 0.0
    if spent_today + planned_cost > daily_cap:
        raise BudgetExceeded(
            f"{engine} daily: spent ${spent_today:.2f} + planned ${planned_cost:.2f} exceeds ${daily_cap:.2f}"
        )

    spent_week = getattr(db, "sum_media_cost_week", lambda e: 0.0)(engine) or 0.0
    if spent_week + planned_cost > weekly_cap:
        raise BudgetExceeded(
            f"{engine} weekly: spent ${spent_week:.2f} + planned ${planned_cost:.2f} exceeds ${weekly_cap:.2f}"
        )

    # Alert at 80%
    alert_threshold = getattr(settings, "fal_alert_threshold", 0.80)
    if spent_week + planned_cost > weekly_cap * alert_threshold:
        log.warning(
            "%s spend at %.0f%% of weekly cap ($%.2f / $%.2f)",
            engine, (spent_week + planned_cost) / weekly_cap * 100,
            spent_week + planned_cost, weekly_cap,
        )

def check_fal_balance(settings: Settings, min_balance: float = 1.0) -> float:
    """Check fal.ai account balance. Raise if below minimum."""
    try:
        resp = httpx.get(
            "https://rest.alpha.fal.ai/balance",
            headers={"Authorization": f"Key {settings.fal_key}"},
            timeout=10,
        )
        resp.raise_for_status()
        balance = float(resp.json().get("balance", 0))
        log.info("fal.ai balance: $%.2f", balance)
        if balance < min_balance:
            raise BudgetExceeded(
                f"fal.ai balance ${balance:.2f} below minimum ${min_balance:.2f}"
            )
        return balance
    except httpx.HTTPError as e:
        log.warning("fal.ai balance check failed: %s", e)
        return float("inf")  # don't block on network errors
```

```python
# aeloria/publishing/tiktok.py
"""TikTok Content Posting API — publish reels as TikTok videos.

Uses the TikTok Content Posting API (v2) for direct video posting.
Requires TikTok Login Kit credentials (client_key, client_secret, access_token, open_id).
"""
import logging
import httpx
from tenacity import retry, stop_after_attempt, wait_exponential, retry_if_exception

log = logging.getLogger(__name__)

TIKTOK_API_BASE = "https://open.tiktokapis.com/v2"

class TikTokError(Exception):
    pass

def _retryable(exc) -> bool:
    if isinstance(exc, (httpx.TimeoutException, httpx.ConnectError)):
        return True
    if isinstance(exc, httpx.HTTPStatusError):
        return exc.response.status_code in {429, 500, 502, 503, 504}
    return False

_RETRY = retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=15, min=15, max=120),
    retry=retry_if_exception(_retryable),
    retry_error_callback=lambda s: s.outcome.result(),
)

@_RETRY
def publish_video(settings, video_url: str, caption: str) -> tuple[str, str]:
    """Publish a video to TikTok. Returns (video_id, share_url)."""
    if not settings.tiktok_access_token:
        raise TikTokError("TikTok not configured — set tiktok_access_token")

    with httpx.Client(timeout=120) as c:
        # Initialize upload — TikTok requires a 2-step: init → upload → publish
        r = c.post(
            f"{TIKTOK_API_BASE}/post/publish/video/init/",
            headers={
                "Authorization": f"Bearer {settings.tiktok_access_token}",
                "Content-Type": "application/json; charset=UTF-8",
            },
            json={
                "post_info": {
                    "title": caption[:150],  # TikTok title limit
                    "privacy_level": "PUBLIC_TO_EVERYONE",
                },
                "source_info": {
                    "source": "PULL_FROM_URL",
                    "video_url": video_url,
                },
            },
        )
        if r.status_code != 200:
            log.error("[tiktok] init failed: %d %s", r.status_code, r.text[:300])
            raise TikTokError(f"TikTok init failed: {r.status_code}")

        data = r.json().get("data", {})
        video_id = str(data.get("video", {}).get("video_id", data.get("publish_id", "")))
        share_url = data.get("share_url", f"https://www.tiktok.com/@unknown/video/{video_id}")

        log.info("[tiktok] video published -> %s", video_id)
        return video_id, share_url
```

### Step 2: Implement repurposing engine

```python
# aeloria/distribution/repurpose.py
"""Content repurposing: reformat one IG post for TikTok, LinkedIn, X, email.

Takes the canonical IG post (image + caption) and produces platform-specific
variants: aspect-ratio reformatted image, platform-tuned caption, and
X thread splits.
"""
import logging
import textwrap
from dataclasses import dataclass, field

log = logging.getLogger(__name__)

# Platform specs
PLATFORM_SPECS = {
    "instagram": {"aspect": "4:5", "caption_limit": 2200, "hashtags": True},
    "tiktok": {"aspect": "9:16", "caption_limit": 150, "hashtags": True},
    "linkedin": {"aspect": "1.91:1", "caption_limit": 3000, "hashtags": False},
    "x": {"aspect": "16:9", "caption_limit": 280, "hashtags": False, "thread": True},
    "email": {"aspect": None, "caption_limit": None, "hashtags": False},
}

@dataclass
class RepurposeResult:
    platforms: dict = field(default_factory=dict)
    # {platform: {"aspect": str, "caption": str|list, "image_url": str|None}}

def reformat_aspect_ratio(source_ratio: str, target_ratio: str) -> str:
    """Return the target aspect ratio (actual reformatting done by fal.ai or PIL)."""
    return target_ratio

def adapt_caption_for_platform(caption: str, platform: str) -> str | list[str]:
    """Adapt caption for platform conventions."""
    spec = PLATFORM_SPECS.get(platform, {})
    limit = spec.get("caption_limit")

    if platform == "linkedin":
        # Strip hashtags, add professional framing
        lines = caption.split("\n")
        clean = " ".join(l for l in lines if not l.strip().startswith("#"))
        return clean.strip()

    if platform == "x" and limit:
        # Split into thread if over 280 chars
        if len(caption) <= limit:
            return [caption]
        sentences = caption.replace("\n", " ").split(". ")
        threads = []
        current = ""
        for s in sentences:
            s = s + "." if not s.endswith(".") else s
            if len(current) + len(s) + 1 <= limit:
                current = f"{current} {s}".strip()
            else:
                if current:
                    threads.append(current)
                current = s
        if current:
            threads.append(current)
        return threads

    if limit and len(caption) > limit:
        return caption[:limit - 3] + "..."

    return caption

def repurpose_post(
    image_url: str | None = None,
    video_url: str | None = None,
    caption: str = "",
    pillar: str = "",
) -> RepurposeResult:
    """Repurpose a canonical IG post for all platforms."""
    result = RepurposeResult()

    for platform, spec in PLATFORM_SPECS.items():
        adapted = adapt_caption_for_platform(caption, platform)

        # Video platforms get video, image platforms get image
        media = video_url if platform in ("tiktok", "instagram") and video_url else image_url

        result.platforms[platform] = {
            "aspect": spec["aspect"],
            "caption": adapted,
            "media_url": media,
            "needs_reformat": spec["aspect"] is not None,
        }

    log.info("Repurposed post for %d platforms", len(result.platforms))
    return result
```

### Step 2: Implement lip-sync module

```python
# aeloria/generation/lipsync.py
"""Lip-sync video generation via fal.ai.

Animates a still/video with synchronized lip movement from an audio track.
Uses fal-ai/wav2lip or equivalent for talking-head reels.
"""
import logging
import httpx

log = logging.getLogger(__name__)

LIPSYNC_MODEL = "fal-ai/wav2lip"
LIPSYNC_COST_USD = 0.15

class LipSyncResult:
    def __init__(self, video_bytes: bytes | None, cost_usd: float = 0.0, gen_params: dict | None = None):
        self.video_bytes = video_bytes
        self.cost_usd = cost_usd
        self.gen_params = gen_params or {}

def lip_sync_video(
    video_url: str,
    audio_url: str | None,
    settings,
    face_detect_boost: bool = True,
    pad_size: int = 10,
) -> LipSyncResult:
    """Generate a lip-synced video from a source video + audio track."""
    if not audio_url:
        log.info("Lip-sync skipped: no audio track provided")
        return LipSyncResult(video_bytes=None, cost_usd=0.0)

    import os
    import fal_client
    os.environ["FAL_KEY"] = settings.fal_key

    arguments = {
        "video_url": video_url,
        "audio_url": audio_url,
        "face_detect_boost": face_detect_boost,
        "pad_size": pad_size,
    }

    log.info("Lip-sync generation: model=%s", LIPSYNC_MODEL)
    try:
        result = fal_client.subscribe(LIPSYNC_MODEL, arguments=arguments)
        video_url_out = result.get("video", {}).get("url")
        if not video_url_out:
            raise RuntimeError(f"Lip-sync returned no video: {result}")

        resp = httpx.get(video_url_out, timeout=120)
        resp.raise_for_status()

        log.info("Lip-sync complete: cost=$%.4f", LIPSYNC_COST_USD)
        return LipSyncResult(
            video_bytes=resp.content,
            cost_usd=LIPSYNC_COST_USD,
            gen_params={"model": LIPSYNC_MODEL, **arguments},
        )
    except Exception as e:
        log.error("Lip-sync failed: %s", e)
        return LipSyncResult(video_bytes=None, cost_usd=0.0)
```
